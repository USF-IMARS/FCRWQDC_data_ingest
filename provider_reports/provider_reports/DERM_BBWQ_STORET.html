# define generic checkcol function
checkColumn <- function(df, colName) {

  if (!colName %in% names(df)) {
    stop(paste("Column", colName, "not found in dataframe"))
  }

  vals <- df[[colName]]
  n <- length(vals)
  n_missing <- sum(is.na(vals))
  pct_missing <- (n_missing / n) * 100

  # --- HEADER ---------------------------------------------------------------
  cat("\n## Column:", colName, "\n")
  cat(sprintf("**Missing values:** %d (%.2f%%)",
              n_missing, pct_missing))

  # --- SHOW TABLE OF MISSING ROWS (QUARTO-FRIENDLY) -------------------------
  if (n_missing > 0) {
    tbl <- head(df[is.na(vals), , drop = FALSE])

    # print() forces Quarto/knitr to display the table even inside functions
    print(
      knitr::kable(
        tbl,
        format = "simple",
        caption = paste("**First rows with", colName, "missing:")
      )
    )

    cat("\n---------------------------------------------------------------------------\n")
  }

  invisible(NULL)
}
