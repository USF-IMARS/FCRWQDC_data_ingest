# define generic checkcol function
checkColumn <- function(df, colName){ 
  if(!colName %in% names(df)){
    stop(paste("Column", colName, "not found in dataframe"))
  }
  
  # Calculate percent missing
  vals <- df[[colName]]
  n <- length(vals)
  n_missing <- sum(is.na(vals))
  pct_missing <- (n_missing / n) * 100
  
  cat(sprintf("\n\n'%s': missing %d values (%.2f%%)\n",
              colName, n_missing, pct_missing))
  
  # If missing, show first few rows with missing
  if(n_missing > 0){
    cat("\tFirst few rows with missing values:\n")
    print(head(df[is.na(vals), , drop = FALSE]))
  }
}
